# QA de la actividad doble: island.exe --qa qa/dual.txt --qa-out <carpeta>
# La canción es lo principal y el temporizador de foco sale en una burbuja
# al lado. Con capturas y comprobaciones; no hay que ejecutarlo a mano.
slow 3
set islandStyle "notch"
set placement "top"
wait-rest
track "Canción" "Artista"
focus start
wait 1200
focus pause
wait-rest
background
grab dual-cerrada
# Con el temporizador en pausa la burbuja está quieta: cero fotogramas.
idle 1000
film dual-abrir 80
open
wait-rest
grab dual-abierta
film dual-cerrar 80
close
wait-rest
grab dual-cerrada2

# Un aviso, un nivel y un toast tapan la burbuja.
notice "Micrófono apagado" "Zoom"
wait-rest
grab dual-aviso
wait 4000
wait-rest
level 0.6
wait-rest
grab dual-nivel
wait 3000
wait-rest
toast "Mensajes" "Ana" "¿Nos vemos a las 6?"
wait-rest
grab dual-toast
wait 6000
wait-rest
background
grab dual-tras-toast
idle 1000

# Sin canción el temporizador pasa a ser lo principal y la burbuja se va.
notrack
wait-rest
background
grab dual-sin-cancion
track "Canción" "Artista"
wait-rest

# Con la opción apagada no hay burbuja.
set dualActivity false
wait-rest
background
grab dual-apagada
